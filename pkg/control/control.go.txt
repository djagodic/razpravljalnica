package server

import (
	"context"

	api "github.com/djagodic/razpravljalnica/pkg/api"
)

type ControlPlaneServer struct {
	api.UnimplementedControlPlaneServer
	head *Node
	tail *Node
}

func NewControlPlaneServer(head, tail *Node) *ControlPlaneServer {
	return &ControlPlaneServer{
		head: head,
		tail: tail,
	}
}

func (c *ControlPlaneServer) GetClusterState(ctx context.Context, _ *emptypb.Empty) (*api.GetClusterStateResponse, error) {
	return &api.GetClusterStateResponse{
		Head: &api.NodeInfo{
			NodeId:  c.head.ID,
			Address: c.head.Address,
		},
		Tail: &api.NodeInfo{
			NodeId:  c.tail.ID,
			Address: c.tail.Address,
		},
	}, nil
}
