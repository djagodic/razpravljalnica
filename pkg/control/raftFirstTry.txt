package control

import (
	"bytes"
	"log"
	"time"

	"github.com/hashicorp/raft"
	"github.com/hashicorp/raft-boltdb"
)

// RaftControlPlane wraps your existing ControlPlaneServer and Raft node
type RaftControlPlane struct {
	*ControlPlaneServer
	raft        *raft.Raft
	raftFSM     *RaftFSM
	raftAddress string // host:port for Raft communication
}

// RaftFSM implements raft.FSM interface
type RaftFSM struct {
	c *ControlPlaneServer
}

// Apply applies a Raft log entry
func (f *RaftFSM) Apply(logEntry *raft.Log) interface{} {
	cmd := logEntry.Data
	// decode command type (RegisterNode, DeregisterNode, etc.)
	// can use simple gob/json
	var rpcCommand RPCCommand
	if err := rpcCommand.Unmarshal(cmd); err != nil {
		log.Printf("failed to unmarshal raft command: %v", err)
		return err
	}

	switch rpcCommand.Op {
	case "RegisterNode":
		f.c.RegisterNode(rpcCommand.Ctx, rpcCommand.RegisterNodeRequest)
	case "DeregisterNode":
		f.c.DeregisterNode(rpcCommand.NodeID)
	}
	return nil
}

func (f *RaftFSM) Snapshot() (raft.FSMSnapshot, error) {
	// TODO: implement snapshot of current nodes for Raft
	return &RaftSnapshot{nodes: f.c.nodes}, nil
}

func (f *RaftFSM) Restore(rc io.ReadCloser) error {
	// TODO: restore snapshot
	return nil
}

// StartRaft initializes the Raft node
func (r *RaftControlPlane) StartRaft(dataDir, raftID string, peers []string) error {
	config := raft.DefaultConfig()
	config.LocalID = raft.ServerID(raftID)

	// in-memory log store for testing, can switch to BoltDB
	store, err := raftboltdb.NewBoltStore(dataDir + "/raft.db")
	if err != nil {
		return err
	}

	snapshots, err := raft.NewFileSnapshotStore(dataDir, 1, log.Default())
	if err != nil {
		return err
	}

	transport, err := raft.NewTCPTransport(r.raftAddress, nil, 3, 10*time.Second, log.Default())
	if err != nil {
		return err
	}

	r.raftFSM = &RaftFSM{c: r.ControlPlaneServer}

	ra, err := raft.NewRaft(config, r.raftFSM, store, store, snapshots, transport)
	if err != nil {
		return err
	}

	r.raft = ra

	// bootstrap cluster if first time
	hasState, err := raft.HasExistingState(store, store, snapshots)
	if err != nil {
		return err
	}
	if !hasState {
		cfg := raft.Configuration{}
		for _, peer := range peers {
			cfg.Servers = append(cfg.Servers, raft.Server{
				ID:      raft.ServerID(peer),
				Address: raft.ServerAddress(peer),
			})
		}
		r.raft.BootstrapCluster(cfg)
	}

	return nil
}
