package main

import (
	"log"
	"net"
	"os"
	"time"

	nadzorna_ravnina "github.com/djagodic/razpravljalnica/pkg/api/nadzornaRavnina"
	control "github.com/djagodic/razpravljalnica/pkg/control"
	"google.golang.org/grpc"
)

type NodeConfig struct {
	ID        string
	Addr      string
	RaftAddr  string
	DataDir   string
	Bootstrap bool
}

func main() {
	nodes := []NodeConfig{
		{ID: "cp1", Addr: "127.0.0.1:5000", RaftAddr: "127.0.0.1:6000", DataDir: "data/cp1", Bootstrap: true},
		{ID: "cp2", Addr: "127.0.0.1:5001", RaftAddr: "127.0.0.1:6001", DataDir: "data/cp2", Bootstrap: false},
		{ID: "cp3", Addr: "127.0.0.1:5002", RaftAddr: "127.0.0.1:6002", DataDir: "data/cp3", Bootstrap: false},
	}

	for _, node := range nodes {
		n := node
		go func() { startNode(n) }()
	}

	select {}
}

func startNode(cfg NodeConfig) {
	if err := os.MkdirAll(cfg.DataDir, 0755); err != nil {
		log.Fatalf("[%s] failed to create data dir: %v", cfg.ID, err)
	}

	// Start gRPC server
	lis, err := net.Listen("tcp", cfg.Addr)
	if err != nil {
		log.Fatalf("[%s] Failed to listen: %v", cfg.ID, err)
	}
	grpcServer := grpc.NewServer()
	cpServer := control.NewControlPlaneServer(cfg.Addr, cfg.RaftAddr)
	raftCP := control.NewRaftControlPlane(cpServer)

	// Register control-plane API
	nadzorna_ravnina.RegisterControlPlaneServer(grpcServer, raftCP)

	// Start Raft with gRPC transport
	if err := raftCP.StartRaftGRPC(cfg.DataDir, cfg.ID, cfg.Bootstrap, grpcServer); err != nil {
		log.Fatalf("[%s] Failed to start Raft: %v", cfg.ID, err)
	}

	log.Printf("[%s] Control Plane running on %s", cfg.ID, cfg.Addr)

	// For non-bootstrap nodes: wait until a leader is found
	if !cfg.Bootstrap {
		go func() {
			for {
				leader := raftCP.GetLeader()
				if leader != "" {
					log.Printf("[%s] Found leader: %s", cfg.ID, leader)
					break
				}
				time.Sleep(500 * time.Millisecond)
			}
		}()
	}

	if err := grpcServer.Serve(lis); err != nil {
		log.Fatalf("[%s] gRPC server failed: %v", cfg.ID, err)
	}
}
