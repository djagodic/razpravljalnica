package main

import (
	"flag"
	"log"
	"net"
	"os"

	nadzorna_ravnina "github.com/djagodic/razpravljalnica/pkg/api/nadzornaRavnina"
	control "github.com/djagodic/razpravljalnica/pkg/control"
	"google.golang.org/grpc"
)

func main() {
	id := flag.String("id", "cp1", "Raft node ID")
	addr := flag.String("addr", "127.0.0.1:5000", "gRPC address")
	raftAddr := flag.String("raft-addr", "127.0.0.1:6000", "Raft transport address")
	dataDir := flag.String("data", "data/cp1", "Raft data directory")
	bootstrap := flag.Bool("bootstrap", false, "Bootstrap this node")
	flag.Parse()

	if err := os.MkdirAll(*dataDir, 0755); err != nil {
		log.Fatalf("[%s] failed to create data dir: %v", *id, err)
	}

	cpServer := control.NewControlPlaneServer(*addr, *raftAddr)
	raftCP := control.NewRaftControlPlane(cpServer)

	grpcServer := grpc.NewServer()

	if err := raftCP.StartRaftGRPC(*dataDir, *id, *bootstrap, *raftAddr, grpcServer); err != nil {
		log.Fatalf("[%s] Failed to start Raft: %v", *id, err)
	}

	lis, err := net.Listen("tcp", *addr)
	if err != nil {
		log.Fatalf("[%s] Failed to listen: %v", *id, err)
	}
	nadzorna_ravnina.RegisterControlPlaneServer(grpcServer, raftCP)
	log.Printf("[%s] Control Plane running on %s", *id, *addr)

	// Serve gRPC
	if err := grpcServer.Serve(lis); err != nil {
		log.Fatalf("[%s] gRPC server failed: %v", *id, err)
	}
}
