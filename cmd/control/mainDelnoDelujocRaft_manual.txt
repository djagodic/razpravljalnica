package main

import (
	"flag"
	"log"
	"strings"

	"net"

	nadzorna_ravnina "github.com/djagodic/razpravljalnica/pkg/api/nadzornaRavnina"
	control "github.com/djagodic/razpravljalnica/pkg/control"
	"google.golang.org/grpc"
)

func main() {
	addr := flag.String("addr", "127.0.0.1:5000", "control plane gRPC address")
	raftAddr := flag.String("raft-addr", "127.0.0.1:6000", "raft transport address")
	bootstrap := flag.Bool("bootstrap", false, "bootstrap raft cluster")


    raftID := flag.String("id", "cp1", "raft node ID")
    peersFlag := flag.String("peers", "127.0.0.1:5000,127.0.0.1:5001,127.0.0.1:5002", "comma-separated peer addresses")
    dataDir := flag.String("data", "data/cp1", "raft data dir")
    flag.Parse()

    peers := []string{}
    for _, p := range splitComma(*peersFlag) {
        peers = append(peers, p)
    }

    // Inicializacija Control Plane + Raft
	cpServer := control.NewControlPlaneServer(*addr, *raftAddr)
    raftCP := control.NewRaftControlPlane(cpServer)
    if err := raftCP.StartRaft(*dataDir, *raftID, peers, *bootstrap); err != nil {
        log.Fatalf("Failed to start Raft: %v", err)
    }

    // gRPC server
    lis, err := net.Listen("tcp", *addr)
    if err != nil {
        log.Fatalf("Failed to listen: %v", err)
    }
    grpcServer := grpc.NewServer()
    nadzorna_ravnina.RegisterControlPlaneServer(grpcServer, raftCP)

    log.Printf("Control Plane running on %s", *addr)
    if err := grpcServer.Serve(lis); err != nil {
        log.Fatalf("gRPC server failed: %v", err)
    }
}

func splitComma(s string) []string {
    return strings.Split(s, ",")
}
