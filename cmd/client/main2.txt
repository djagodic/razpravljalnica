package main

import (
	"context"
	"fmt"
	"log"
	"os"
	"time"

	api "github.com/djagodic/razpravljalnica/pkg/api"
	control "github.com/djagodic/razpravljalnica/pkg/control"
	"google.golang.org/grpc"
)

func getClusterState(controlAddr string) (*api.NodeInfo, *api.NodeInfo, error) {
	conn, err := grpc.Dial(controlAddr, grpc.WithInsecure())
	if err != nil {
		return nil, nil, err
	}
	defer conn.Close()

	ctrlClient := control.NewControlPlaneClient(conn)
	ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
	defer cancel()

	resp, err := ctrlClient.GetClusterState(ctx, &api.Empty{})
	if err != nil {
		return nil, nil, err
	}

	return resp.Head, resp.Tail, nil
}

func main() {
	if len(os.Args) < 2 {
		fmt.Println("usage: client <control_plane_addr>")
		return
	}

	controlAddr := os.Args[1]

	// pridobi head/tail node
	head, tail, err := getClusterState(controlAddr)
	if err != nil {
		log.Fatalf("failed to get cluster state: %v", err)
	}
	fmt.Printf("Head node: %s (%s), Tail node: %s (%s)\n", head.NodeId, head.Address, tail.NodeId, tail.Address)

	// poveži se na head za pisalne operacije
	conn, err := grpc.Dial(head.Address, grpc.WithInsecure())
	if err != nil {
		log.Fatalf("failed to connect to head: %v", err)
	}
	defer conn.Close()
	client := api.NewMessageBoardClient(conn)

	ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
	defer cancel()

	// primer: ustvari uporabnika
	u, err := client.CreateUser(ctx, &api.CreateUserRequest{Name: "David"})
	if err != nil {
		log.Fatalf("create user error: %v", err)
	}
	fmt.Println("Created user:", u.Id, u.Name)

	// ustvari temo
	t, err := client.CreateTopic(ctx, &api.CreateTopicRequest{Name: "Tema1"})
	if err != nil {
		log.Fatalf("create topic error: %v", err)
	}
	fmt.Println("Created topic:", t.Id, t.Name)

	// pošlji sporočilo
	m, err := client.PostMessage(ctx, &api.PostMessageRequest{
		TopicId: t.Id, UserId: u.Id, Text: "Pozdrav, razpravljalnica!"})
	if err != nil {
		log.Fatalf("post message error: %v", err)
	}
	fmt.Println("Posted message id:", m.Id, m.Text)

	// subscribe stream
	streamCtx, streamCancel := context.WithCancel(context.Background())
	defer streamCancel()

	// dobi subscribe node in token
	subResp, err := client.GetSubcscriptionNode(ctx, &api.SubscriptionNodeRequest{
		UserId: u.Id, TopicId: []int64{t.Id}})
	if err != nil {
		log.Fatalf("get subscription node error: %v", err)
	}

	subConn, err := grpc.Dial(subResp.Node.Address, grpc.WithInsecure())
	if err != nil {
		log.Fatalf("failed to connect to subscription node: %v", err)
	}
	defer subConn.Close()

	subClient := api.NewMessageBoardClient(subConn)
	subStream, err := subClient.SubscribeTopic(streamCtx, &api.SubscribeTopicRequest{
		UserId: u.Id, TopicId: []int64{t.Id}, FromMessageId: 0,
		SubscribeToken: subResp.SubscribeToken,
	})
	if err != nil {
		log.Fatalf("subscribe failed: %v", err)
	}

	go func() {
		for {
			ev, err := subStream.Recv()
			if err != nil {
				log.Printf("subscription ended: %v", err)
				return
			}
			fmt.Printf("[EVENT] %v: %s\n", ev.Op, ev.Message.Text)
		}
	}()

	// block main goroutine, da vidimo dogodke
	fmt.Println("Listening to topic events for 15s...")
	time.Sleep(15 * time.Second)
	fmt.Println("Exiting client")
}
