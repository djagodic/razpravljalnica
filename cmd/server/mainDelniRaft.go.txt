package main

import (
	"context"
	"flag"
	"log"
	"net"
	"strings"
	"time"

	nadzorna_ravnina "github.com/djagodic/razpravljalnica/pkg/api/nadzornaRavnina"
	razpravljalnica "github.com/djagodic/razpravljalnica/pkg/api/razpravljalnica"
	"github.com/djagodic/razpravljalnica/pkg/server"
	"google.golang.org/grpc"
	"google.golang.org/grpc/credentials/insecure"
)

// ------------------------ Heartbeat ------------------------
func startHeartbeat(cpAddr, nodeID string) {
    conn, err := grpc.NewClient(cpAddr, grpc.WithTransportCredentials(insecure.NewCredentials()))
    if err != nil {
        log.Fatalf("heartbeat dial failed: %v", err)
    }
    defer conn.Close()
    client := nadzorna_ravnina.NewControlPlaneClient(conn)

    ticker := time.NewTicker(2 * time.Second)
    for range ticker.C {
        _, err := client.Heartbeat(context.Background(), &nadzorna_ravnina.HeartbeatRequest{NodeId: nodeID})
        if err != nil {
            log.Printf("heartbeat failed: %v", err)
        }
    }
}

// ------------------------ Register Node ------------------------
func registerNode(nodeID, addr string, peers []string) *nadzorna_ravnina.RegisterNodeResponse {
    req := &nadzorna_ravnina.RegisterNodeRequest{
        NodeId:  nodeID,
        Address: addr,
    }

    for _, peer := range peers {
        conn, err := grpc.NewClient(peer, grpc.WithTransportCredentials(insecure.NewCredentials()))
        if err != nil {
            log.Printf("dial failed %s: %v", peer, err)
            continue
        }
        client := nadzorna_ravnina.NewControlPlaneClient(conn)
        resp, err := client.RegisterNode(context.Background(), req)
        conn.Close()
        if err == nil {
            log.Printf("Registered at %s: %+v", peer, resp)
            return resp
        }
        log.Printf("Failed at %s: %v, trying next", peer, err)
    }
    log.Fatalf("Failed to register node at any peer")
    return nil
}

func main() {
    addr := flag.String("addr", ":50051", "node gRPC address")
    nodeID := flag.String("id", "node-1", "node ID")
    peers := flag.String("peers", "127.0.0.1:5000,127.0.0.1:5001,127.0.0.1:5002", "control plane peers")
    flag.Parse()

    peerList := []string{}
    for _, p := range splitComma(*peers) {
        peerList = append(peerList, p)
    }

    // Registracija node
    resp := registerNode(*nodeID, *addr, peerList)
    isHead := resp.IsHead
    isTail := resp.IsTail

    // Heartbeat
    go startHeartbeat(peerList[0], *nodeID)

    // gRPC server za board
    s := grpc.NewServer()
    board := server.NewMessageBoardServer(*nodeID, isHead, isTail)
    razpravljalnica.RegisterMessageBoardServer(s, board)
    board.StartSubscribingChanges(*nodeID, nadzorna_ravnina.NewControlPlaneClient(nil)) // <-- tu uporabi ctrlClient

    lis, err := net.Listen("tcp", *addr)
    if err != nil {
        log.Fatalf("listen failed: %v", err)
    }
    log.Printf("Node %s listening at %s (head=%v, tail=%v)", *nodeID, *addr, isHead, isTail)
    if err := s.Serve(lis); err != nil {
        log.Fatalf("serve failed: %v", err)
    }
}

func splitComma(s string) []string {
    return strings.Split(s, ",")
}
